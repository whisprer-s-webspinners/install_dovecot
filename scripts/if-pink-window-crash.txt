# 1. Kill frozen terminals (Ctrl+C in each)
# If stuck, from working terminal:
sudo pkill -9 apt
sudo pkill -9 dpkg

# 2. Fix broken package state
sudo dpkg --configure -a
sudo apt --fix-broken install

# 3. Run cleanup script to start fresh
chmod +x cleanup_mailserver.sh
sudo ./cleanup_mailserver.sh

# 4. Use NEW fixed install script
chmod +x install_mailserver_fixed.sh
sudo ./install_mailserver_fixed.sh
